\Heading
{Financial Literacy}
{Supplement 5. Find Rate}
{\textbf{Instructions: }For each problem, sketch a timeline, write an equation
    in one variable, and solve for that variable to answer the question.}

\begin{questions}

    \mathproblem[][3in]{
        Determine the nominal rate compounded quarterly at which \$2,000 will accumulate to
        \$3,000 in 3 years 9 months.
    }

    \mathproblem[][3in]{
        Determine the nominal rate compounded quarterly at which \$2,000 will accumulate to
        \$3,000 in 3 years 9 months.
    }

    \mathproblem[][2in]{
        José deposited \$1,000 three and a half years ago.  He has \$1,581.72 in his account
        now.  What rate of interest compounded semiannually did he earn on his deposit?
    }

    \mathproblem[][2in]{
        The population of a county grew at a constant compound rate from 200,000 at the beginning of
        2000 to 275,000 at the beginning of 2020.  How many people were added to the
        population between the beginning of 2005 and the beginning of 2015?
    }

    \mathproblem[][2in]{
        An investment fund advertises that they can triple your money in 10 years.  What rate of
        interest compounded monthly are they implying?
    }

    \mathproblem[][2in]{
    Find the annual effective rate equivalent to

        \bpoint 16\% compounded quarterly?
        \bpoint 18\% compounded monthly
        \bpoint 9.25\% compounded daily
}

    \mathproblem[][2in]{
If \$3,000 grows to \$4,000 over 10 years, what nominal annual rate, compounded
    monthly, did it earn?
    }

\end{questions}
