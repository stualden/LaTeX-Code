\Heading
{Financial Literacy}
{Supplement 1. Discounted Value}
{\textbf{Instructions: }For each problem, sketch a timeline, write an equation
    in one variable, and solve for that variable to answer the question.}

\begin{questions}

    \mathproblem[][3in]{
        On her 20th birthday, Sally received \$10,000 from an account into which her parents made a deposit
        on her 5th birthday.  How much was that deposit, if the account earns 6\% compounded semiannually
        (i.e., twice a year)?
    }

    \mathproblem[][3in]{
        You can buy a building lot (a) for \$180,000 in cash now, or (b) for payments of \$100,000 now,
        \$50,000 at 1 year from now, and \$50,000 at 2 years from now.  Which arrangement is better for
        you, if money is worth 8\% compounded monthly?
    }

    \mathproblem[][3in]{
        If there are currently 25,000 people in Smallville, and the population has been growing at a rate of 6\%
        per year, how many people lived in Smallville 12 years ago?
    }

    \mathproblem[][3in]{
        You need to buy a new refrigerator.  The appliance store gives you the option pay in full 6 months from now,
        or to pay now and receive a 1\% discount off the purchase price.  Which arrangement will you choose, if
        you're able to invest surplus cash in an account earning 2.75\% compounded monthly?
    }


\end{questions}
