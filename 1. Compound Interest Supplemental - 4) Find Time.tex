\Heading
{Financial Literacy}
{Supplement 4. Find Time}
{\textbf{Instructions: }For each problem, sketch a timeline, write an equation
    in one variable, and solve for that variable to answer the question.}

\begin{questions}

    \mathproblem[][3in]{
        How long will it take \$2,000 to earn \$800 in interest if it earns 10\% compounded
        quarterly?
    }

    \mathproblem[][3in]{
        If the cost of living rises 8\% per year, how long will it take for the
        purchasing power of \$1 to fall to 60\textcent?
    }

    \mathproblem[][3in]{
        How long does it take for an investment to double in value if it earns 15\% compounded
        daily?
    }

    \mathproblem[][3in]{
        If the population of the world doubles every 25 years, how long does it take to
        increase by  50\%?
    }

\end{questions}
