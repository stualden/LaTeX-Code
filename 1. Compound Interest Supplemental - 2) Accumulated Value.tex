\Heading
{Financial Literacy}
{Supplement 2. Accumulated Value}
{\textbf{Instructions: }For each problem, sketch a timeline, write an equation
    in one variable, and solve for that variable to answer the question.}

\begin{questions}

    \mathproblem[][3in]{
        In 1492, Queen Isabella sponsored Christopher Columbus's journey by giving him \$10,000.  If
        instead she had put the money into a bank account paying 3\% annual interest and let it grow,
        how much money would be in the account in 2024?
    }

    \mathproblem[][3in]{
        If Julie puts \$250 into an account earning 9\% compounded monthly, how much interest will
        she earn in the first year?  How much interest will she earn in the third year?
    }

    \mathproblem[][3in]{
        The population of the U.S. in 2021 was about 330 million.  If the population grows at an
        annual rate of 2.3\%, how many people will live in the U.S. in 2050?
    }

    \mathproblem[][3in]{
        You have access to an account which pays 7\% compounded monthly.  If you deposit \$1,000
        now and then another \$1,000 a year from now and another \$1,000 two years from now, how
        much will you have at the end of 3 years?
    }

\end{questions}
