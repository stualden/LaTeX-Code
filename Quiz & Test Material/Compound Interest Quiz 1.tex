\Heading
{Financial Literacy}
{Compound Interest Quiz 1}
{20 Minutes, 100 points.
  No aids except a calculator.
  Answer each question by selecting the best answer.
Show all mathematical work; timeline sketches recommended.}

\begin{questions}

  \multichoice[20][1in]
  {
    (Accumulated Value) On the day he was born, Sam's parents put \$1,000 into a bank account
    earning 6\% compounded monthly.  How much money will Sam be able to withdraw from the account
    on his 18th birthday?
  }
  {
    \choice \$2,945.33
    \CorrectChoice \$2,936.77
    \choice \$2,854.34
    \choice \$2,963.77
  }

  \multichoice[20][.5in]
  {
    (Nominal Rates) 10\% compounded quarterly is equivalent to X\% compounded monthly.
    Find X.
  }
  {
    \choice 9.29\%
    \CorrectChoice 9.92\%
    \choice 10.02\%
    \choice 9.57\%
  }

  \multichoice[20][.75in]
  {
    (Discounted Value) If you earn 8\% compounded monthly, and need \$10,000 for a car in 3
    years, how much do you need to set aside today?
  }
  {
    \choice \$7,728.55
    \choice \$7,912.34
    \CorrectChoice \$7,782.55
    \choice \$7938.32
  }

  \multichoice[20][.75in]
  {
    (Find Rate) If your \$25 grows to \$45 in 7 years, what was your nominal interest rate,
    compounded monthly?
  }
  {
    \choice 8.67\%
    \choice 8.34\%
    \CorrectChoice 8.43\%
    \choice 8.76\%
  }

  \multichoice[20][.75in]
  {
    % \begin{mcquestion}[20]
    (Find Time) How many years does it take \$40 to grow to \$75 at 8\% compounded quarterly?
  }
  {
    \choice 7.7 years
    \CorrectChoice 7.9 years
    \choice 8.1 years
    \choice 8.5 years
  }

\end{questions}
