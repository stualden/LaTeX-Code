  \begin{problem}[][1.5in]{You just lent a friend \$300 and told him you'd like him to repay you
      someday at 7\% annual interest.  How much does he owe you if he pays
    you back two years from now?}
  \end{problem}

  \begin{problem}[][1.5in]{Five (5) years ago, you put some money in a savings account earning 3.5\%
    interest.  You now have \$118.77. How much did you put in the account 5 years ago?}
  \end{problem}

  \begin{problem}[][1.5in]{Same situation as in problem 1, but your friend pays you back in only
    6 months.  How much does he owe you then?}
  \end{problem}

  \begin{problem}[][2in]{You buy shares in a mutual fund for \$150.  After 6 years, you sell
    your shares for \$212.78. What annual rate of interest did you earn?}
  \end{problem}

  \begin{problem}[][2in]{You buy \$1,000 worth of stock in a company whose stock price is
      growing at a rate of 10\% per year.  How many years must you wait
    before the stock is worth \$1,250?}
  \end{problem}

  \begin{problem}[][2in]{At 5\% annual interest, how long does it take for your money to triple?}
  \end{problem}

\end{questions}
